function doPost(e) {
  try {
    // v2.3: Stripe webhook — its payload has no `action`; routed by URL key.
    if (e && e.parameter && e.parameter.hook) return handleStripeHook(e);
    const payload = JSON.parse(e.postData.contents);
    const action  = payload.action || '';

    if (action === 'submit')        return handleSubmit(payload);
    if (action === 'get_market')    return handleGetMarket(payload);
    if (action === 'get_counts')    return handleGetCounts();
    if (action === 'delete_token')  return handleDeleteToken(payload);
    if (action === 'ai_identify')   return handleAiIdentify(payload);   // v1.3
    if (action === 'ai_identify2')  return handleAiIdentify2(payload);  // v2.4
    if (action === 'ai_verify_photo') return handleAiVerifyPhoto(payload); // v2.7
    if (action === 'ref_photo')     return handleRefPhoto(payload);     // v2.8
    if (action === 'ai_card')       return handleAiCard(payload);       // v2.2
    if (action === 'sub_check')     return handleSubCheck(payload);     // v2.3
    if (action === 'ai_quota')      return handleAiQuota(payload);      // v3.3
    if (action === 'import_map')    return handleImportMap(payload);    // v3.4
    if (action === 'barcode_pair')  return handleBarcodePair(payload); // v3.6

    return respond(400, 'Unknown action');
  } catch(err) {
    return respond(500, 'Server error: ' + err.message);
  }
}

function doGet(e) {
  try {
    return handleGetCounts();
  } catch(err) {
    return respond(500, 'Server error: ' + err.message);
  }
}


// ============================================================
//  ACTION: ai_quota (v3.3) — "how many photo IDs do I have left?"
//  Payload: { action:'ai_quota', token }
//  Returns: { status:200, remaining, cap } — premium-aware, spends nothing.
//  Powers the app's always-on "X of Y reads left today" counter (v1472).
// ============================================================
function handleAiQuota(payload) {
  const token = String(payload.token || '').trim();
  if (!token) return respond(400, 'Missing token');
  const config = getConfig();
  const cap = _ai2CapFor(token, config);
  let used = 0;
  try {
    const sheet = getSheet(AI_USAGE_TAB);
    const today = _aiToday();
    const data  = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === token && _aiDateStr(data[i][1]) === today) {
        used = parseInt(data[i][2]) || 0;
        break;
      }
    }
  } catch (e) {}
  return respond(200, 'OK', { remaining: Math.max(0, cap - used), cap: cap });
}

// ============================================================
//  ACTION: ai_identify  (v1.3)
//  Payload: { action:'ai_identify', token, image (base64, no data: prefix),
//             mime ('image/jpeg'), hints:{ scale, type, mfrs:[] } }
//  Returns: { status:200, text:'Manufacturer SKU...', remaining, cached }
//           { status:429, message:'quota' }  when the device cap is hit
// ============================================================

function handleAiIdentify(payload) {
  // ── validate ──
  if (!payload.token) return respond(400, 'Missing token');
  const token = String(payload.token).replace(/[^a-z0-9]/gi, '').substring(0, 16);
  if (token.length < 6) return respond(400, 'Invalid token');

  const img = String(payload.image || '');
  if (!img) return respond(400, 'Missing image');
  if (img.length > AI_MAX_IMAGE_B64) return respond(400, 'Image too large');
  // base64 sanity (also blocks accidental "data:image/..." prefixes)
  if (!/^[A-Za-z0-9+/=]+$/.test(img.substring(0, 100))) return respond(400, 'Bad image encoding');

  const mime = (payload.mime === 'image/png') ? 'image/png' : 'image/jpeg';

  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) return respond(500, 'AI not configured');

  // ── duplicate-photo cache (retry taps cost nothing) ──
  // v1.4: prompt version is part of the cache key — a prompt change must not
  // serve answers cached under the old format.
  const hash = _aiHash(img + '|' + AI_PROMPT_VER);
  const cached = _aiCacheGet(hash);
  if (cached) {
    return respond(200, 'OK', { text: cached, cached: true, remaining: _aiRemaining(token) });
  }

  // ── per-device daily cap ──
  const config = getConfig();
  const cap = parseInt(config.ai_daily_cap) || AI_DAILY_CAP_DEFAULT;
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  let used;
  try {
    used = _aiUsageIncrement(token, cap);
  } finally {
    lock.releaseLock();
  }
  if (used < 0) {
    return respond(429, 'quota', { remaining: 0, cap: cap });
  }

  // ── build the prompt (mirrors the app's Lens prompt + labeled format) ──
  const hints = payload.hints || {};
  let subject = [String(hints.scale || ''), String(hints.type || '')].filter(String).join(' ').trim()
             || 'model train item';
  let mfrs = Array.isArray(hints.mfrs) ? hints.mfrs.filter(String).slice(0, 5) : [];
  const mfrPhrase = mfrs.length ? ', possibly made by ' + mfrs.join(' or ') : '';

  const prompt =
      'This is a photo of a toy/model train collectible (' + subject + mfrPhrase + '): a train box or '
    + 'box-end label, the model/train item itself, a BUILDING or accessory, or a PAPER item '
    + '(poster, catalog, instruction sheet, brochure, or other advertising). '
    + 'IMPORTANT — the photo may show a shelf or table with MANY items. Identify ONLY the central, '
    + 'most prominent, in-focus item. Numbers printed on OTHER items or boxes near it are NOT this '
    + 'item\'s number — ignore them. '
    + 'If it is the model itself (no box), read the lettering painted/printed ON the model. '
    + 'If it is a paper item, read the title and any form/part number printed on it. '
    + 'Read the printed text and identify the item. Respond with EXACTLY these lines, '
    + 'each on its own line, nothing else:\n'
    + 'Manufacturer SKU or catalog number: <the unique product/catalog code printed on the '
    + 'item, e.g. 6-22993, 30-4021, 6464-125, 279-4324. This is NOT the cab/road number painted on the '
    + 'model. If none is printed, SEARCH THE WEB for this exact product (use its appearance, lettering, '
    + 'brand, and type) and give the catalog number from a confident match. If still not certain, '
    + 'write exactly: no specific catalog number found>\n'
    + 'Manufacturer: <brand, e.g. Lionel, MTH, Atlas, Weaver, K-Line, Menards, RMT, or Unknown>\n'
    + 'Year manufactured: <year manufactured or published. If you are not certain but can estimate from search results or knowledge, prefix with ~ like ~2021. Write Unknown ONLY if you cannot even estimate>\n'
    + 'Scale or gauge: <e.g. O, O-27, Standard Gauge, HO, S, or Unknown>\n'
    + 'Description: <one short line describing the item as printed on it>\n'
    + 'If it is a locomotive or train car, ALSO add these lines:\n'
    + 'Road name: <the railroad represented, e.g. Santa Fe, or Unknown>\n'
    + 'Cab number: <the number painted on the model, or Unknown>\n'
    + 'Locomotive class or body style: <e.g. F3, Boxcar, GP-9, or Unknown>\n'
    + 'If it is a building or accessory, ALSO add this line:\n'
    + 'Structure type: <what the building/accessory is, e.g. Passenger Station, Auto Dealership, Water Tower>\n'
    + 'If it is a paper or advertising item, ALSO add these lines:\n'
    + 'Title: <the title printed on it>\n'
    + 'Form or part number: <the form/part number printed on it, or Unknown>\n'
    + 'Belongs to item or set: <which product, set, or catalog number it goes with, or Unknown>\n'
    + 'Original or reproduction: <Original, Reproduction, or Unknown>\n'
    + 'THEN, only if you confidently recognize this exact product (catalog number + brand) '
    + 'from your general knowledge, ALSO add any of these lines — omit any you are not sure of:\n'
    + 'Known year: <year(s) this product was made; prefix with ~ if approximate, e.g. ~2021>\n'
    + 'Known road name: <railroad represented, if any>\n'
    + 'Known number: <the number painted on the model>\n'
    + 'Known body style: <e.g. Caboose, Boxcar, F3>\n'
    + 'Known scale or gauge: <e.g. O, O-27, Standard Gauge, HO>\n'
    + 'Known description: <one line about the product — INCLUDE the product line/series name (e.g. Tinplate Traditions, RailKing, Postwar Celebration), color, features>\n'
    + 'Rules: you have Google Search available. If no catalog number is printed on the item, you MUST '
    + 'run at least one Google search for it before answering (search its brand, lettering, colors, and '
    + 'type — e.g. "Menards O gauge power plant building"). Do not answer from the photo alone in that case. '
    + 'When the search identifies the product, the Description must name the specific product, not just '
    + 'the building type. Never invent a catalog number: it must be '
    + 'printed on the item OR come from a confident web match for this exact product. Never repeat the '
    + 'cab number as the SKU. Apart from a searched catalog number, the first section reports what is '
    + 'printed on the item; the Known lines carry your general knowledge and search findings, and only '
    + 'when the product identification is certain. '
    + 'When you DO recognize the product, always include at least Known year and Known scale or gauge. '
    + 'Always try to give a year: an approximate ~year from search findings beats Unknown.';

  // ── call Gemini ──
  const model = PropertiesService.getScriptProperties().getProperty('GEMINI_MODEL') || AI_MODEL_DEFAULT;
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + model
            + ':generateContent?key=' + encodeURIComponent(key);
  const body = {
    contents: [{
      parts: [
        { inline_data: { mime_type: mime, data: img } },
        { text: prompt },
      ],
    }],
    tools: [{ google_search: {} }],
    // v3.1 (2026-08-03): thinkingBudget 1024 -> 0. Same measurement as v3.0's
    // identify2 change: thinking was the bulk of billed output and the answers
    // did not need it. handleAiCard has run this model with thinkingBudget 0
    // since v2.2.1. v1 is only reachable as identify2's fallback, but a
    // fallback that costs 4x is still a leak.
    generationConfig: { temperature: 0.1, maxOutputTokens: 2048, thinkingConfig: { thinkingBudget: 0 } },
  };

  let res;
  try {
    res = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(body),
      muteHttpExceptions: true,
    });
  } catch (err) {
    _aiUsageRefund(token);   // network failure shouldn't burn a scan
    return respond(502, 'AI unreachable: ' + err.message);
  }

  const code = res.getResponseCode();
  if (code !== 200) {
    _aiUsageRefund(token);
    // 429 from Google = the whole project's free tier is exhausted today.
    if (code === 429) return respond(503, 'AI busy — try again later');
    return respond(502, 'AI error HTTP ' + code);
  }

  let text = '';
  try {
    const j = JSON.parse(res.getContentText());
    const parts = j.candidates && j.candidates[0] && j.candidates[0].content
               && j.candidates[0].content.parts;
    text = (parts || []).map(function(p) { return p.text || ''; }).join('\n').trim();
  } catch (err) {
    return respond(502, 'AI answer unreadable');
  }
  if (!text) return respond(502, 'AI gave no answer');
  text = text.substring(0, 2000);

  _aiCachePut(hash, text);

  return respond(200, 'OK', { text: text, cached: false, remaining: Math.max(0, cap - used) });
}

  // Collect images: preferred `images` array, else v1-style single image.
  let rawImages = [];
  if (Array.isArray(payload.images) && payload.images.length) {
    rawImages = payload.images;
  } else if (payload.image) {
    rawImages = [{ data: payload.image, mime: payload.mime }];
  }
  if (!rawImages.length) return respond(400, 'Missing image');
  rawImages = rawImages.slice(0, AI2_MAX_IMAGES);

  const images = [];
  for (let i = 0; i < rawImages.length; i++) {
    const data = String((rawImages[i] && rawImages[i].data) || '');
    if (!data) return respond(400, 'Empty image #' + (i + 1));

    if (!/^[A-Za-z0-9+/=]+$/.test(data.substring(0, 100))) return respond(400, 'Bad image encoding #' + (i + 1));
    const mime = ((rawImages[i] && rawImages[i].mime) === 'image/png') ? 'image/png' : 'image/jpeg';

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  let used, g;
  try {
    used = _aiUsageIncrement(token, cap);
    // v2.6 GUARD RAIL: global monthly ceiling for the grounded v2 pool.

    : (PropertiesService.getScriptProperties().getProperty('GEMINI_MODEL2') || AI2_MODEL_DEFAULT);
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + model
            + ':generateContent?key=' + encodeURIComponent(key);
  const parts = images.map(function(im) {
    return { inline_data: { mime_type: im.mime, data: im.data } };
  });
  parts.push({ text: prompt });
  const body = {
    contents: [{ parts: parts }],
    tools: [{ google_search: {} }],
    generationConfig: { temperature: 0.1, maxOutputTokens: 4096,
                        thinkingConfig: { thinkingLevel: 'minimal' } },
  };

  let res;
  try {
    res = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(body),
      muteHttpExceptions: true,
    });
  } catch (err) {
    _aiUsageRefund(token);   // network failure shouldn't burn a scan
    if (g > 0) _ai2GlobalRefund();   // v2.6: give the month its call back too
    return respond(502, 'AI unreachable: ' + err.message);
  }

  const code = res.getResponseCode();
  if (code !== 200) {
    _aiUsageRefund(token);
    if (g > 0) _ai2GlobalRefund();   // v2.6
    // 429 from Google = the whole project's free tier is exhausted today.
    if (code === 429) return respond(503, 'AI busy — try again later');
    return respond(502, 'AI error HTTP ' + code);
  }

  let text = '';
  try {
    const j = JSON.parse(res.getContentText());
    const p = j.candidates && j.candidates[0] && j.candidates[0].content
           && j.candidates[0].content.parts;
    text = (p || []).map(function(x) { return x.text || ''; }).join('\n').trim();
  } catch (err) {
    return respond(502, 'AI answer unreadable');
  }
  if (!text) return respond(502, 'AI gave no answer');
  text = text.substring(0, 2000);

  _aiCachePut(hash, text);

  return respond(200, 'OK', { text: text, cached: false, remaining: Math.max(0, cap - used) });
}



// ============================================================
//  ACTION: ref_photo  (v2.8 — Brad's free side-by-side compare)
//  Fetches the master row's Reference Link page and returns its product
//  photo so the APP can show it next to the collector's own photo and let
//  the HUMAN judge. No Gemini call, no daily-pool spend, no image upload.
//  Payload: { action:'ref_photo', token, refUrl }
//  Returns: { status:200, img:<base64>, mime, refImg:<image url>, cached }
//           { status:422, message:'no reference image' }
// ============================================================

function handleRefPhoto(payload) {
  if (!payload.token) return respond(400, 'Missing token');
  const token = String(payload.token).replace(/[^a-z0-9]/gi, '').substring(0, 16);
  if (token.length < 6) return respond(400, 'Invalid token');
  const refUrl = String(payload.refUrl || '').substring(0, 400);
  if (!/^https?:\/\//i.test(refUrl)) return respond(400, 'Missing or bad refUrl');
  // v2.9: '#ANCHOR' marks the item's spot on multi-item pages — split it off
  // before fetching, hand it to the extractor.
  const hashAt = refUrl.indexOf('#');
  const pageUrl = hashAt >= 0 ? refUrl.substring(0, hashAt) : refUrl;
  const anchor = hashAt >= 0 ? refUrl.substring(hashAt + 1) : '';

  // Cache by FULL URL (anchor included — different items, different photos).
  const cacheKey = 'rp_' + _aiHash(refUrl);
  try {
    const hit = CacheService.getScriptCache().get(cacheKey);
    if (hit) {
      const p = JSON.parse(hit);
      return respond(200, 'OK', { img: p.i, mime: p.m, refImg: p.u, cached: true });
    }
  } catch (e) {}

  // Fetch the product page and find its photo (anchor-aware, v2.9).
  let refImgUrl = '';
  try {
    const page = UrlFetchApp.fetch(pageUrl, { muteHttpExceptions: true, followRedirects: true });
    if (page.getResponseCode() === 200) refImgUrl = _vpFindImage(page.getContentText(), pageUrl, anchor);
  } catch (err) { /* handled below */ }
  if (!refImgUrl) return respond(422, 'no reference image');

  // Download the photo itself.
  let refB64 = '', refMime = 'image/jpeg';
  try {
    const imgRes = UrlFetchApp.fetch(refImgUrl, { muteHttpExceptions: true, followRedirects: true });
    if (imgRes.getResponseCode() !== 200) throw new Error('HTTP ' + imgRes.getResponseCode());
    const blob = imgRes.getBlob();
    const bytes = blob.getBytes();
    if (bytes.length < 1000 || bytes.length > AI3_MAX_REF_BYTES) throw new Error('bad size ' + bytes.length);
    const ct = String(blob.getContentType() || '');
    if (ct.indexOf('image/') !== 0) throw new Error('not an image: ' + ct);
    refMime = (ct === 'image/png') ? 'image/png' : 'image/jpeg';
    refB64 = Utilities.base64Encode(bytes);
  } catch (err) { return respond(422, 'no reference image'); }

  try {
    const packed = JSON.stringify({ i: refB64, m: refMime, u: refImgUrl });
    if (packed.length < 95000) CacheService.getScriptCache().put(cacheKey, packed, 21600);
  } catch (e) {}
  return respond(200, 'OK', { img: refB64, mime: refMime, refImg: refImgUrl, cached: false });
}

// v2.7 helper: pull the product photo URL out of a product page's HTML.
// Tries og:image / twitter:image / link image_src, then the first large-looking
// content <img>. Returns '' when nothing usable is found.
function _vpFindImage(htmlText, baseUrl, anchor) {
  // v2.9: anchor first — on multi-item archive pages (COTT) the right photo
  // is the first real image AFTER the item's id="ANCHOR" marker.
  if (anchor) {
    const esc = String(anchor).replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
    const am = htmlText.match(new RegExp('(?:id|name)=["\']' + esc + '["\']', 'i'));
    if (am) {
      const after = htmlText.substring(am.index);
      const hit = _vpContentImg(after, baseUrl);
      if (hit) return hit;
    }
  }
  const metas = [
    /<meta[^>]+property=["']og:image(?::secure_url)?["'][^>]+content=["']([^"']+)["']/i,
    /<meta[^>]+content=["']([^"']+)["'][^>]+property=["']og:image(?::secure_url)?["']/i,
    /<meta[^>]+name=["']twitter:image["'][^>]+content=["']([^"']+)["']/i,
    /<link[^>]+rel=["']image_src["'][^>]+href=["']([^"']+)["']/i,
  ];
  for (let i = 0; i < metas.length; i++) {
    const m = htmlText.match(metas[i]);
    if (m && m[1]) return _vpAbsUrl(m[1], baseUrl);
  }
  return _vpContentImg(htmlText, baseUrl);
}

// v2.9: first <img> that plausibly shows a PRODUCT. Two passes — real
// upload-library photos first (WordPress /wp-content/uploads/), then any
// non-chrome image. Theme/social chrome is never returned (the old code
// grabbed COTT's YouTube icon, which then flunked the size check → 422).
function _vpContentImg(htmlText, baseUrl) {
  const imgs = htmlText.match(/<img[^>]+src=["']([^"']+)["'][^>]*>/gi) || [];
  const chrome = /logo|icon|sprite|banner|header|footer|avatar|pixel|blank|spacer|button|\/themes\/|\/socials\//i;
  for (let pass = 0; pass < 2; pass++) {
    for (let i = 0; i < imgs.length; i++) {
      const m = imgs[i].match(/src=["']([^"']+)["']/i);
      if (!m) continue;
      const u = m[1];
      if (chrome.test(u)) continue;
      if (!/\.(jpe?g|png|webp)(\?|$)/i.test(u)) continue;
      if (pass === 0 && !/wp-content\/uploads\//i.test(u)) continue;
      return _vpAbsUrl(u, baseUrl);
    }
  }
  return '';
}

function _vpAbsUrl(u, baseUrl) {
  u = String(u || '').trim().replace(/&amp;/g, '&');   // pages HTML-escape query strings
  if (!u) return '';
  if (/^https?:\/\//i.test(u)) return u;
  const m = baseUrl.match(/^(https?:\/\/[^\/]+)(\/[^?#]*)?/i);
  if (!m) return '';
  if (u.indexOf('//') === 0) return 'https:' + u;
  if (u.indexOf('/') === 0) return m[1] + u;
  const dir = (m[2] || '/').replace(/[^\/]*$/, '');
  return m[1] + dir + u;
}


// ── AI helpers ──────────────────────────────────────────────

function _aiHash(b64) {
  const raw = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, b64);
  return raw.map(function(b) {
    const v = (b < 0 ? b + 256 : b).toString(16);
    return v.length === 1 ? '0' + v : v;
  }).join('');
}

function _aiToday() {
  return Utilities.formatDate(new Date(), 'America/New_York', 'yyyy-MM-dd');
}

// v1.3.1: Sheets auto-converts "2026-07-03" text into a real Date cell,
// so comparing the cell to a date STRING silently never matches (that
// broke the daily cap). Normalize whatever the cell holds back to text.
function _aiDateStr(v) {
  if (v instanceof Date) return Utilities.formatDate(v, 'America/New_York', 'yyyy-MM-dd');
  return String(v || '');
}

// Increment today's counter for this device. Returns the new count,
// or -1 if the cap was already reached (nothing incremented).
function _aiUsageIncrement(token, cap) {
  const sheet = getSheet(AI_USAGE_TAB);
  const today = _aiToday();
  const data  = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === token && _aiDateStr(data[i][1]) === today) {
      const count = parseInt(data[i][2]) || 0;
      if (count >= cap) return -1;
      sheet.getRange(i + 1, 3).setValue(count + 1);
      return count + 1;
    }
  }
  sheet.appendRow([token, today, 1]);
  // Light housekeeping: drop rows older than ~30 days once the tab gets big.
  if (data.length > 2000) _aiUsagePurge(sheet);
  return 1;
}

// Give a scan back when the AI call itself failed (network/Google error).
function _aiUsageRefund(token) {
  try {
    const sheet = getSheet(AI_USAGE_TAB);
    const today = _aiToday();
    const data  = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === token && _aiDateStr(data[i][1]) === today) {
        const count = parseInt(data[i][2]) || 0;
        if (count > 0) sheet.getRange(i + 1, 3).setValue(count - 1);
        return;
      }
    }
  } catch (e) {}
}

function _aiRemaining(token) {
  try {
    const cap = parseInt(getConfig().ai_daily_cap) || AI_DAILY_CAP_DEFAULT;
    const sheet = getSheet(AI_USAGE_TAB);
    const today = _aiToday();
    const data  = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === token && _aiDateStr(data[i][1]) === today) {
        return Math.max(0, cap - (parseInt(data[i][2]) || 0));
      }
    }
    return cap;
  } catch (e) { return 0; }
}

function _aiUsagePurge(sheet) {
  try {
    const cutoff = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000);
    const cutStr = Utilities.formatDate(cutoff, 'America/New_York', 'yyyy-MM-dd');
    const data = sheet.getDataRange().getValues();
    for (let i = data.length - 1; i >= 1; i--) {
      if (_aiDateStr(data[i][1]) < cutStr) sheet.deleteRow(i + 1);
    }
  } catch (e) {}
}

function _aiCacheGet(hash) {
  try {
    const sheet = getSheet(AI_CACHE_TAB);
    const data  = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === hash) {
        sheet.getRange(i + 1, 4).setValue((parseInt(data[i][3]) || 0) + 1);
        return String(data[i][1] || '');
      }
    }
  } catch (e) {}
  return null;
}

function _aiCachePut(hash, text) {
  try {
    const sheet = getSheet(AI_CACHE_TAB);
    sheet.appendRow([hash, text, new Date().toISOString(), 0]);
    // Keep the cache tab from growing forever — oldest rows go first.
    if (sheet.getLastRow() > 3000) {
      sheet.deleteRows(2, 500);
    }
  } catch (e) {}
}

// ============================================================
//  ACTION: submit
// ============================================================

function handleSubmit(payload) {
  if (!payload.token)  return respond(400, 'Missing token');
  if (!payload.items || !payload.items.length) return respond(400, 'No items');

  const token = String(payload.token).replace(/[^a-z0-9]/gi, '').substring(0, 16);
  if (token.length < 6) return respond(400, 'Invalid token');

  const sheet = getSheet(SUBMISSIONS_TAB);
  const now   = new Date().toISOString();
  const rows  = [];

  for (const item of payload.items) {
    if (!item.item_num)   continue;

    // v1.2: in_master flag — defaults to TRUE for older app versions
    // that don't send it, so legacy submissions behave exactly as before.
    const inMaster = !(item.in_master === false || item.in_master === 'false');

    // Normalize condition — accept both "8" and "C8" formats
    let condRaw = String(item.condition || '').toUpperCase().trim();
    if (/^\d+$/.test(condRaw)) condRaw = 'C' + condRaw;  // "8" → "C8"
    if (!VALID_CONDITIONS.includes(condRaw)) {
      // v1.2: market rows still require a valid condition; a not-in-master
      // row is kept anyway (blank condition) so the catalog review sees it.
      if (inMaster) continue;
      condRaw = '';
    }

    const itemNum      = String(item.item_num).replace(/[^a-zA-Z0-9\-\.]/g, '').substring(0, 20);
    const variation    = String(item.variation || '').substring(0, 60);
    const estWorth     = parseFloat(item.est_worth)  || 0;
    const soldPrice    = parseFloat(item.sold_price) || 0;
    // v1.2 extras — harmless blanks for older app versions.
    const manufacturer = String(item.manufacturer || '').substring(0, 30);
    const description  = String(item.description  || '').substring(0, 140);
    const roadName     = String(item.road_name    || '').substring(0, 60);
    const source       = String(item.source       || '').substring(0, 20);

    rows.push([token, itemNum, variation, condRaw, estWorth, soldPrice, now,
               inMaster, manufacturer, description, roadName, source]);
  }

  if (!rows.length) return respond(400, 'No valid items after validation');

  removeRowsByToken(sheet, token);

  const lastRow = sheet.getLastRow();
  sheet.getRange(lastRow + 1, 1, rows.length, 12).setValues(rows);

  rebuildComputed();

  return respond(200, 'OK', { rows_written: rows.length });
}

function handleGetMarket(payload) {
  if (!payload.item_num) return respond(400, 'Missing item_num');

  const config    = getConfig();

  const itemNum   = String(payload.item_num).replace(/[^a-zA-Z0-9\-\.]/g, '');
  const variation = String(payload.variation || '');

  const computed = getComputedData();
  const key      = makeKey(itemNum, variation);
  const entry    = computed[key];

  if (!entry) return respond(200, 'no_data', { reason: 'not_found' });

  const result = {};
  for (const [cond, data] of Object.entries(entry)) {
    if (data.count >= minPoints) {
      result[cond] = {
        avg:   Math.round(data.avg),
        low:   Math.round(data.low),
        high:  Math.round(data.high),
        count: data.count
      };
    }
  }

  if (!Object.keys(result).length) {
    return respond(200, 'no_data', { reason: 'below_threshold', threshold: minPoints });
  }

  return respond(200, 'OK', { market: result });
}

// ============================================================
//  COMPUTED TAB BUILDER
// ============================================================

function rebuildComputed() {
  const config  = getConfig();
  const trimPct = (config.trim_percent || 10) / 100;
  const sheet   = getSheet(SUBMISSIONS_TAB);
  const data    = sheet.getDataRange().getValues();

  const groups = {};
  for (let i = 1; i < data.length; i++) {
    const row       = data[i];
    const itemNum   = row[1];
    const variation = row[2];
    const condition = row[3];
    const estWorth  = parseFloat(row[4]) || 0;
    if (!itemNum || !condition || estWorth <= 0) continue;

    const key = makeKey(itemNum, variation) + '|' + condition;
    if (!groups[key]) groups[key] = { itemNum, variation, condition, values: [] };
    groups[key].values.push(estWorth);
  }

  const outputRows = [['item_num','variation','condition','avg','low','high','count','updated']];
  const now = new Date().toISOString();

  for (const [key, g] of Object.entries(groups)) {
    const trimmed = trimOutliers(g.values, trimPct);
    if (!trimmed.length) continue;

    const avg  = trimmed.reduce((a,b) => a+b, 0) / trimmed.length;
    const low  = Math.min(...trimmed);
    const high = Math.max(...trimmed);

    outputRows.push([g.itemNum, g.variation, g.condition,
                     Math.round(avg), Math.round(low), Math.round(high),
                     g.values.length, now]);
  }

  const computedSheet = getSheet(COMPUTED_TAB);
  computedSheet.clearContents();
  computedSheet.getRange(1, 1, outputRows.length, 8).setValues(outputRows);
}

function getRarityLabel(itemCount, totalContributors, config) {
  if (!totalContributors || totalContributors < 1) return null;
  const pct = (itemCount / totalContributors) * 100;

  const bands = [
    config.rare_band_1 || 0.5,
    config.rare_band_2 || 3,
    config.rare_band_3 || 10,
    config.rare_band_4 || 30,
    config.rare_band_5 || 60
  ];

  if (pct < bands[0]) return 'Highly Rare';
  if (pct < bands[1]) return 'Rare';
  if (pct < bands[2]) return 'Hard to Find';
  if (pct < bands[3]) return 'Scarce';
  if (pct < bands[4]) return 'Plentiful';
  return 'Common';
}

function getSheet(name) {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(name);
  if (!sheet) throw new Error('Tab not found: ' + name);
  return sheet;
}

function getConfig() {
  const sheet  = getSheet(CONFIG_TAB);
  const data   = sheet.getDataRange().getValues();
  const config = {};
  for (let i = 1; i < data.length; i++) {
    const key = String(data[i][0]).trim();
    const val = data[i][1];
    if (key) config[key] = isNaN(val) ? val : parseFloat(val);
  }
  return config;
}

function getComputedData() {
  const sheet = getSheet(COMPUTED_TAB);
  const data  = sheet.getDataRange().getValues();
  const result = {};
  for (let i = 1; i < data.length; i++) {
    const [itemNum, variation, condition, avg, low, high, count] = data[i];
    if (!itemNum) continue;
    const key = makeKey(itemNum, variation);
    if (!result[key]) result[key] = {};
    result[key][condition] = { avg, low, high, count };
  }
  return result;
}

// ── generic utilities (restored from the v1.2 archive — the v1.3 project-
// folder copy was truncated exactly here; unchanged since v1.2) ──
function makeKey(itemNum, variation) {
  return String(itemNum).trim() + '|' + String(variation || '').trim();
}

function removeRowsByToken(sheet, token) {
  const data = sheet.getDataRange().getValues();
  let removed = 0;
  for (let i = data.length - 1; i >= 1; i--) {
    if (data[i][0] === token) {
      sheet.deleteRow(i + 1);
      removed++;
    }
  }
  return removed;
}

function trimOutliers(values, trimPct) {
  if (values.length < 3) return values;
  const sorted   = [...values].sort((a, b) => a - b);
  const cutCount = Math.floor(sorted.length * trimPct);
  return sorted.slice(cutCount, sorted.length - cutCount);
}

function respond(status, message, data) {
  const payload = { status, message, ...(data || {}) };
  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}


// ============================================================
//  ACTION: ai_card  (v2.2) — business-card reader for Contacts
//  Payload: { action:'ai_card', token, image (base64, no data: prefix), mime }
//  Returns: { status:200, text:'Name: ...\nTitle: ...', remaining, cached }
//           { status:429, message:'quota' } when the shared daily cap is hit
//  Shares ai_daily_cap + the ai_cache tab with ai_identify.
// ============================================================
const AI_CARD_PROMPT_VER = 'c2';   // bump on every card-prompt change (c2: kills answers cached under the truncating c1 config)

function handleAiCard(payload) {
  if (!payload.token) return respond(400, 'Missing token');
  const token = String(payload.token).replace(/[^a-z0-9]/gi, '').substring(0, 16);
  if (token.length < 6) return respond(400, 'Invalid token');

  const img = String(payload.image || '');
  if (!img) return respond(400, 'Missing image');
  if (img.length > AI_MAX_IMAGE_B64) return respond(400, 'Image too large');
  if (!/^[A-Za-z0-9+/=]+$/.test(img.substring(0, 100))) return respond(400, 'Bad image encoding');

  const mime = (payload.mime === 'image/png') ? 'image/png' : 'image/jpeg';

  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) return respond(500, 'AI not configured');

  // duplicate-photo cache (re-scanning the same card costs nothing)
  const hash = _aiHash(img + '|card|' + AI_CARD_PROMPT_VER);
  const cached = _aiCacheGet(hash);
  if (cached) {
    return respond(200, 'OK', { text: cached, cached: true, remaining: _aiRemaining(token) });
  }

  // shared per-device daily cap
  const config = getConfig();
  const cap = parseInt(config.ai_daily_cap) || AI_DAILY_CAP_DEFAULT;
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  let used;
  try {
    used = _aiUsageIncrement(token, cap);
  } finally {
    lock.releaseLock();
  }
  if (used < 0) {
    return respond(429, 'quota', { remaining: 0, cap: cap });
  }

  const prompt =
      'This is a photo of a BUSINESS CARD. Transcribe what is printed on it. '
    + 'Respond with EXACTLY these labeled lines, each on its own line, nothing else. '
    + 'If something is not on the card, leave that line blank after the colon.\n'
    + 'Name: <the person\'s name>\n'
    + 'Title: <their job title, e.g. Owner, Retail Store Manager>\n'
    + 'Business: <the company or store name, including LLC/Inc if printed — read the logo if that is where the name is>\n'
    + 'Store phone: <the main / store / office number>\n'
    + 'Cell phone: <the number marked c, cell, or mobile>\n'
    + 'Home phone: <the number marked h or home>\n'
    + 'Email: <the email address exactly as printed>\n'
    + 'Website: <the web address, without http:// or www.>\n'
    + 'Address: <street, city, state zip — the FIRST address if the card lists several>\n'
    + 'Rules: transcribe EXACTLY what is printed — never guess missing digits or letters. '
    + 'A lone unlabeled phone number goes on the Store phone line. OMIT fax numbers entirely. '
    + 'Read 1/l and 0/O carefully in emails and web addresses (domains are normally lowercase '
    + 'words or numbers), and if the email domain and website are clearly the same, spell them '
    + 'the same. No commentary, no extra lines.';

  const model = PropertiesService.getScriptProperties().getProperty('GEMINI_MODEL') || AI_MODEL_DEFAULT;
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + model
            + ':generateContent?key=' + encodeURIComponent(key);
  const body = {
    contents: [{
      parts: [
        { inline_data: { mime_type: mime, data: img } },
        { text: prompt },
      ],
    }],
    // v2.2.1: thinking OFF — transcription needs none, and Gemini's default
    // thinking tokens count against maxOutputTokens; with 512 the reply was
    // cut off after ~3 lines (Brad's George LaHaye scan: no phones/email).
    generationConfig: { temperature: 0, maxOutputTokens: 1024, thinkingConfig: { thinkingBudget: 0 } },
  };

  let res;
  try {
    res = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(body),
      muteHttpExceptions: true,
    });
  } catch (err) {
    _aiUsageRefund(token);
    return respond(502, 'AI unreachable: ' + err.message);
  }

  const code = res.getResponseCode();
  if (code !== 200) {
    _aiUsageRefund(token);
    if (code === 429) return respond(503, 'AI busy — try again later');
    return respond(502, 'AI error HTTP ' + code);
  }

  let text = '';
  try {
    const j = JSON.parse(res.getContentText());
    const parts = j.candidates && j.candidates[0] && j.candidates[0].content
               && j.candidates[0].content.parts;
    text = (parts || []).map(function(p) { return p.text || ''; }).join('\n').trim();
  } catch (err) {
    return respond(502, 'AI answer unreadable');
  }
  if (!text) return respond(502, 'AI gave no answer');
  text = text.substring(0, 1200);

  _aiCachePut(hash, text);

  return respond(200, 'OK', { text: text, cached: false, remaining: Math.max(0, cap - used) });
}


// ============================================================
//  ACTION: barcode_pair  (v3.6) — community UPC→item pairings
//  Payload: { action:'barcode_pair', token, pairs:[{u,n,m,im,h,d}] }
//    u = upc digits · n = item number · m = maker · im = 1/0 in-master
//    h = how ('scan'…) · d = YYYY-MM-DD first seen on that device
// ============================================================

function handleBarcodePair(payload) {
  if (!payload.token) return respond(400, 'Missing token');
  const token = String(payload.token).replace(/[^a-z0-9]/gi, '').substring(0, 16);
  if (token.length < 6) return respond(400, 'Invalid token');
  const pairs = payload.pairs;
  if (!pairs || !pairs.length) return respond(400, 'No pairs');
  if (pairs.length > 200) return respond(400, 'Too many pairs');

  const sh = _subSheet_(BARCODE_PAIRS_TAB,
    ['upc', 'item_num', 'mfr', 'in_master', 'how', 'first_seen', 'last_seen', 'report_count', 'status']);
  const data = sh.getDataRange().getValues();
  const byKey = {};
  for (let i = 1; i < data.length; i++) {
    byKey[String(data[i][0]) + '|' + String(data[i][1])] = i + 1;   // sheet row
  }
  const now = new Date().toISOString().slice(0, 10);
  let added = 0, bumped = 0;
  for (const p of pairs) {
    const upc = String(p.u || '').replace(/\D/g, '').substring(0, 14);
    const num = String(p.n || '').replace(/[^a-zA-Z0-9\-\.]/g, '').substring(0, 20);
    if (upc.length < 8 || !num) continue;                 // not a UPC / not a number
    const key = upc + '|' + num;
    const row = byKey[key];
    if (row) {
      // a repeat report: bump the count, stamp last_seen — "how many
      // independent people saw this pairing" is what review reads.
      const cur = Number(sh.getRange(row, 8).getValue()) || 1;
      sh.getRange(row, 7, 1, 2).setValues([[now, cur + 1]]);
      bumped++;
    } else {
      const mfr = String(p.m || '').substring(0, 30);
      const how = String(p.h || 'scan').substring(0, 20);
      const first = String(p.d || now).substring(0, 10);
      sh.appendRow([upc, num, mfr, p.im ? true : false, how, first, now, 1, 'new']);
      byKey[key] = sh.getLastRow();
      added++;
    }
  }
  // ok:true is REQUIRED — the app's _bcPairDrain clears its queue only on
  // r.ok || r.success || r.status==='ok', and respond() sends numeric status.
  return respond(200, 'ok', { ok: true, added: added, repeats: bumped });
}

function _subSheet_(name, headers) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(name);
  if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); }
  return sh;
}
function _subNorm_(em) { return String(em || '').trim().toLowerCase(); }
function _subIso_(d) { return Utilities.formatDate(d, 'UTC', 'yyyy-MM-dd'); }
function _subCfg_(key, dflt) {
  try {
    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG_TAB);
    if (!sh) return dflt;
    const vals = sh.getDataRange().getValues();
    for (let i = 0; i < vals.length; i++) {
      if (String(vals[i][0]).trim() === key) return String(vals[i][1]).trim();
    }
  } catch (err) {}
  return dflt;
}
// Find a row by email in col A. Returns {row, vals} or null.
function _subFind_(sheetName, email) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  if (!sh) return null;
  const vals = sh.getDataRange().getValues();
  for (let i = 1; i < vals.length; i++) {
    if (_subNorm_(vals[i][0]) === email) return { row: i + 1, vals: vals[i], sheet: sh };
  }
  return null;
}

// ── ACTION: sub_check  {action:'sub_check', email} ──
// Answers: {status:200, sub:'beta'|'active'|'trial'|'expired', daysLeft,
//           paidThrough, freeUntil, payLink, enforce}
// Fail-open philosophy lives in the APP (network failure = full access);
// this endpoint just reports the truth.
function handleSubCheck(p) {
  const email = _subNorm_(p.email);
  if (!email || email.indexOf('@') < 0) return respond(400, 'bad email');
  _subSheet_(SUBSCRIBERS_TAB, ['email', 'stripe_customer', 'paid_through', 'status', 'updated_at']);
  _subSheet_(TRIALS_TAB, ['email', 'trial_start']);
  _subSheet_(BETA_TAB, ['email', 'free_until', 'note']);
  const today = new Date();
  // ── v3.5: beta tracking (auto-enroll + last-seen/version stamp) ──
  // Guarded end to end: any failure here must never break the sub answer.
  try {
    const wantCode = PropertiesService.getScriptProperties().getProperty('BETA_CODE') || 'BETA2026';
    const gotCode = String(p.betaCode || '').trim().toUpperCase();
    if (gotCode && gotCode === wantCode && !_subFind_(BETA_TAB, email)) {
      _subSheet_(BETA_TAB, ['email', 'free_until', 'note'])
        .appendRow([email, '', 'auto-enrolled ' + _subIso_(today)]);
    }
  } catch (eEnroll) {}
  try {
    const cacheLS = CacheService.getScriptCache();
    const ckLS = 'blseen_' + email;
    if (!cacheLS.get(ckLS)) {
      const bRow = _subFind_(BETA_TAB, email);
      if (bRow) {
        bRow.sheet.getRange(bRow.row, 4).setValue(_subIso_(today));
        const av = String(p.appVersion || '').substring(0, 20);
        if (av) bRow.sheet.getRange(bRow.row, 5).setValue(av);
        cacheLS.put(ckLS, '1', 21600);
      }
    }
  } catch (eSeen) {}
  const base = { payLink: _subCfg_('pay_link', ''), enforce: _subCfg_('sub_enforce', 'off') === 'on' };

  // 1. Beta testers — free year (or open-ended while free_until is blank).
  const beta = _subFind_(BETA_TAB, email);
  if (beta) {
    const fu = beta.vals[1] ? new Date(beta.vals[1]) : null;
    if (!fu || fu.getTime() >= today.getTime()) {
      return respond(200, 'ok', Object.assign(base, { sub: 'beta', freeUntil: beta.vals[1] ? _subIso_(fu) : '' }));
    }
  }
  // 2. Paying subscribers (grace window for card retries).
  const sub = _subFind_(SUBSCRIBERS_TAB, email);
  if (sub && sub.vals[2]) {
    const pt = new Date(sub.vals[2]);
    if (pt.getTime() + SUB_GRACE_DAYS * 864e5 >= today.getTime()) {
      return respond(200, 'ok', Object.assign(base, { sub: 'active', paidThrough: _subIso_(pt) }));
    }
  }
  // 3. Trial — starts silently on first sight of this email.
  let tr = _subFind_(TRIALS_TAB, email);
  if (!tr) {
    _subSheet_(TRIALS_TAB, ['email', 'trial_start']).appendRow([email, _subIso_(today)]);
    return respond(200, 'ok', Object.assign(base, { sub: 'trial', daysLeft: TRIAL_DAYS }));
  }
  const started = new Date(tr.vals[1]);
  const used = Math.floor((today.getTime() - started.getTime()) / 864e5);
  if (used < TRIAL_DAYS) {
    return respond(200, 'ok', Object.assign(base, { sub: 'trial', daysLeft: TRIAL_DAYS - used }));
  }
  return respond(200, 'ok', Object.assign(base, { sub: 'expired', daysLeft: 0 }));
}

// ── STRIPE WEBHOOK  (URL: <script-url>?hook=<WEBHOOK_KEY script property>) ──
// Events to select in Stripe: checkout.session.completed, invoice.paid,
// customer.subscription.deleted. Keeps `subscribers` current automatically.
function handleStripeHook(e) {
  const want = PropertiesService.getScriptProperties().getProperty('WEBHOOK_KEY') || '';
  if (!want || String(e.parameter.hook) !== want) return respond(403, 'bad hook key');
  let ev;
  try { ev = JSON.parse(e.postData.contents); } catch (err) { return respond(400, 'bad payload'); }
  const type = ev.type || '';
  const obj = (ev.data && ev.data.object) || {};
  const sh = _subSheet_(SUBSCRIBERS_TAB, ['email', 'stripe_customer', 'paid_through', 'status', 'updated_at']);
  const now = new Date();
  const yearOut = new Date(now.getTime() + 366 * 864e5);

  function upsert(email, customer, paidThrough, status) {
    email = _subNorm_(email);
    if (!email) return;
    const hit = _subFind_(SUBSCRIBERS_TAB, email);
    const rowVals = [email, customer || (hit ? hit.vals[1] : ''), paidThrough ? _subIso_(paidThrough) : (hit ? String(hit.vals[2]) : ''), status, _subIso_(now)];
    if (hit) hit.sheet.getRange(hit.row, 1, 1, 5).setValues([rowVals]);
    else sh.appendRow(rowVals);
  }

  if (type === 'checkout.session.completed') {
    const email = (obj.customer_details && obj.customer_details.email) || obj.customer_email || '';
    upsert(email, obj.customer || '', yearOut, 'active');
  } else if (type === 'invoice.paid' || type === 'invoice.payment_succeeded') {
    upsert(obj.customer_email || '', obj.customer || '', yearOut, 'active');
  } else if (type === 'customer.subscription.deleted') {
    // No email on this event — match by customer id; keep paid_through
    // (access runs to the end of the paid year), mark canceled.
    const cust = String(obj.customer || '');
    if (cust) {
      const vals = sh.getDataRange().getValues();
      for (let i = 1; i < vals.length; i++) {
        if (String(vals[i][1]) === cust) {
          sh.getRange(i + 1, 4).setValue('canceled');
          sh.getRange(i + 1, 5).setValue(_subIso_(now));
          break;
        }
      }
    }
  }
  return respond(200, 'ok');
}

function handleImportMap(payload) {
  // ── validate ──
  if (!payload.token) return respond(400, 'Missing token');

  try {

  } catch (e) {}

  // ── key: the SECOND (billing-enabled) key; reader key untouched ──
  const props = PropertiesService.getScriptProperties();
  const key = props.getProperty('IMPORT_GEMINI_KEY') || props.getProperty('GEMINI_API_KEY');
  if (!key) return respond(500, 'No API key configured');
  const model = props.getProperty('IMPORT_GEMINI_MODEL') || 'gemini-3.6-flash';

  // ── prompt ──
  const fields = (data.appFields && data.appFields.length) ? data.appFields : [
    'itemNum','manufacturer','gauge','yourDesc','rawGrade','location',
    'priceItem','userEstWorth','yearMade','notes','roadName','roadNumber',
    'hasBox','datePurchased','purchasedFrom','quantity','ignore'];
  const prompt =
      'You are helping a model-train collector import their personal inventory spreadsheet '
    + 'into a collection app. You receive each sheet tab\'s name, its column headers, its row '
    + 'count, and up to 10 sample rows (some flagged highlighted=true when the row has a '
    + 'colored fill; fillGroups lists each distinct highlight color and how many rows use it).\n\n'
    + 'Answer with STRICT JSON only — no prose, no markdown fences — in exactly this shape:\n'
    + '{"mappings": {"<tabName>": {"<header>": "<field>"}},\n'
    + ' "tabClass": {"<tabName>": "trains|books|vehicles|other"},\n'
    + ' "gradeTable": [{"raw": "<their grade string>", "condition": "<1-10 or empty>"}],\n'
    + ' "questions": [{"id": "<short-id>", "text": "<question>", "options": ["<opt>", ...]}]}\n\n'
    + 'Rules:\n'
    + '- Allowed field values, ONLY these: ' + fields.join(', ') + '\n'
    + '- itemNum is the manufacturer catalog number. yourDesc is their free-text description. '
    +   'rawGrade is their condition/grade column VERBATIM (do not convert it in mappings). '
    +   'priceItem is what they PAID; userEstWorth is what they think it is WORTH now. '
    +   'Map a header to "ignore" only when it clearly should not import. Headers you cannot '
    +   'place: omit them (the app asks the user).\n'
    + '- Map EVERY tab that has headers, including non-train tabs (books, die-cast vehicles, '
    +   'airplanes) — classify those in tabClass so the app imports them as manual entries.\n'
    + '- gradeTable: list every DISTINCT grade string you can see in the samples and your best '
    +   '1-10 app-condition guess for each (e.g. "C10/P10" -> "10", "C7" -> "7"). Two grading '
    +   'systems in one cell (like C10/P10): guess from the FIRST. Unsure: empty string.\n'
    + '- questions: up to 5, ONLY about things visible in the data that genuinely need the '
    +   'user\'s answer. If fillGroups shows highlighted rows, the FIRST question must ask what '
    +   'that highlight means, with options ["For sale","Sold","Needs repair","Just formatting","Something else"]. '
    +   'Other good questions: an ambiguous column meaning, a grading system you could not read, '
    +   'a column with person names. Each question needs 2-6 short options. Fewer real '
    +   'questions is better than 5 padded ones.\n\n'
    + 'THE DATA:\n' + dataStr;

  const body = {
    contents: [{ parts: [{ text: prompt }] }],
    generationConfig: { temperature: 0.1, maxOutputTokens: 8192,
                        responseMimeType: 'application/json',
                        thinkingConfig: { thinkingLevel: 'minimal' } },
  };

  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + model
            + ':generateContent?key=' + encodeURIComponent(key);
  let res;
  try {
    res = UrlFetchApp.fetch(url, {
      method: 'post', contentType: 'application/json',
      payload: JSON.stringify(body), muteHttpExceptions: true,
    });
  } catch (e) { return respond(503, 'AI unreachable'); }

  const code = res.getResponseCode();
  if (code === 429 || code === 503) return respond(503, 'AI busy');
  if (code !== 200) return respond(500, 'AI error ' + code);

  let answer = null;
  try {
    const j = JSON.parse(res.getContentText());
    const parts = j.candidates && j.candidates[0] && j.candidates[0].content
               && j.candidates[0].content.parts;
    const text = (parts || []).map(function(p) { return p.text || ''; }).join('').trim();
    answer = JSON.parse(text.replace(/^```json\s*/i, '').replace(/```\s*$/, ''));
  } catch (e) { return respond(500, 'AI answer unparseable'); }

  return respond(200, 'OK', { answer: answer, model: model });
}

// Monday digest — plain-text email to DIGEST_EMAIL (Script Property,
// default support@therailroster.com). Also runnable by hand any time.
function betaWeeklyDigest() {
  const to = PropertiesService.getScriptProperties().getProperty('DIGEST_EMAIL') || 'support@therailroster.com';
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(BETA_TAB);
  if (!sh) return;
  const vals = sh.getDataRange().getValues();
  const now = new Date();
  const d7 = new Date(now.getTime() - 7 * 864e5);
  const d14 = new Date(now.getTime() - 14 * 864e5);
  const newbies = [], active = [], quiet = [];
  for (let i = 1; i < vals.length; i++) {
    const email = String(vals[i][0] || '').trim();
    if (!email) continue;
    const note = String(vals[i][2] || '');
    const rawSeen = vals[i][3];
    const lastSeen = rawSeen ? new Date(rawSeen) : null;
    const seenOk = lastSeen && !isNaN(lastSeen.getTime());
    const ver = String(vals[i][4] || '');
    const name = String(vals[i][5] || '');
    const label = (name ? name + ' <' + email + '>' : email) + (ver ? ' — ' + ver : '');
    const em = note.match(/auto-enrolled (\d{4}-\d{2}-\d{2})/);
    if (em && new Date(em[1]).getTime() >= d7.getTime()) newbies.push(label);
    if (seenOk && lastSeen.getTime() >= d7.getTime()) {
      active.push(label + ' (' + _subIso_(lastSeen) + ')');
    } else if (!seenOk || lastSeen.getTime() < d14.getTime()) {
      quiet.push(label + (seenOk ? ' (last ' + _subIso_(lastSeen) + ')' : ' (never signed in)'));
    }
  }
  const total = vals.length - 1;
  const lines = [
    'The Rail Roster — beta digest, ' + _subIso_(now), '',
    'Testers on the roster: ' + total, '',
    'New this week (' + newbies.length + '):',
    newbies.length ? '  ' + newbies.join('\n  ') : '  none', '',
    'Active in the last 7 days (' + active.length + '):',
    active.length ? '  ' + active.join('\n  ') : '  none', '',
    'Quiet 14+ days or never signed in (' + quiet.length + '):',
    quiet.length ? '  ' + quiet.join('\n  ') : '  none', '',
    '(Sent automatically every Monday by the relay — betaWeeklyDigest, v3.5.)',
  ];
  MailApp.sendEmail(to,
    'Rail Roster beta: ' + active.length + ' active, ' + newbies.length + ' new, ' + quiet.length + ' quiet',
    lines.join('\n'));
}
